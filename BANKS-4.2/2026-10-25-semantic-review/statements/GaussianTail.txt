noncomputable def carrierVar (D t : ℝ) (hD : 0 < D) (ht : 0 < t) : NNReal :=
  ⟨2 * D * t, by positivity⟩

    (gaussianReal (u * t) (carrierVar D t hD ht)).real {x | x ≤ r}
