def localRun (initial : Nat → Bool) : Nat → Nat → Bool
  | 0, site => initial site
  | n+1, 0 => localRun initial n 0
  | n+1, site+1 => localRun initial n site

def healthy : Nat → Bool := fun _ => false
def attacked : Nat → Bool := fun site => decide (site=0)
