noncomputable def escrowStep (E r w h : ℝ) : ℝ := if w ≤ E + r then E + r - w else E + r - h

noncomputable def escrow (E₀ : ℝ) (r w h : ℕ → ℝ) : ℕ → ℝ
  | 0 => E₀
  | k + 1 => escrowStep (escrow E₀ r w h k) (r k) (w k) (h k)
