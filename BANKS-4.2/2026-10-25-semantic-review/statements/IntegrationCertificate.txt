def sumB : List (ℚ × ℚ) → ℚ
  | [] => 0
  | (_, b) :: tail => b + sumB tail

def intervalOK : List (ℚ × ℚ) → ℚ → ℚ → Prop
  | [], base, D => base ≤ D
  | (r, b) :: tail, base, D =>
      max base r + b + sumB tail ≤ D ∧ intervalOK tail (max base r + b) D
