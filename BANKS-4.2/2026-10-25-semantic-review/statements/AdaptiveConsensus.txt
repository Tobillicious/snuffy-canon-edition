def evolution {I : Type} [Fintype I]
    (weights : Nat → I → I → ℝ) (measurement : Nat → I → ℝ)
    (alpha : ℝ) (initial : I → ℝ) : Nat → I → ℝ
  | 0, i => initial i
  | n+1, i => (1-alpha)*(∑ j, weights n i j * evolution weights measurement alpha initial n j)
      +alpha*measurement n i
