noncomputable def outVar (v : ℝ≥0) : ℝ≥0 :=
  ⟨g ^ 2 * ∫ x, (x - ∫ y, y ∂PX) ^ 2 ∂PX + v,
    add_nonneg (mul_nonneg (sq_nonneg g)
      (integral_nonneg (f := fun x : ℝ => (x - ∫ y, y ∂PX) ^ 2) fun _ => sq_nonneg _)) v.2⟩
