noncomputable def qmi {ρ : Matrix (α × β) (α × β) ℂ} (hρ : ρ.IsHermitian) : ℝ :=
  vnEnt (ptrA_isHermitian hρ) + vnEnt (ptrB_isHermitian hρ) - vnEnt hρ

    (hρ' : (controlled W * ρ * star (controlled W)).IsHermitian) :
