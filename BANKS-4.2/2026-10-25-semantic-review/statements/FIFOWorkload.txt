namespace SnuffyFIFO

structure Job where
  release : Nat
  budget : Nat
  deadline : Nat
  deriving Repr

def finish (base : Nat) (jobs : List Job) (work : Job → Nat) : Nat :=
  match jobs with
  | [] => base
  | j :: tail => finish (max base j.release + work j) tail work
