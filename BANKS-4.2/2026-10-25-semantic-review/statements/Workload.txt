def finish (F₀ : ℝ) (r b : ℕ → ℝ) : ℕ → ℝ
  | 0 => F₀
  | i + 1 => max (r (i + 1)) (finish F₀ r b i) + b (i + 1)

def serial (start lead : ℕ → ℝ) (τ : ℕ → ℕ → ℝ) : ℕ → ℕ → ℝ
  | i, 0 => start i
  | 0, k + 1 => lead (k + 1)
  | i + 1, k + 1 => max (serial start lead τ (i + 1) k) (serial start lead τ i (k + 1))
      + τ (i + 1) (k + 1)
