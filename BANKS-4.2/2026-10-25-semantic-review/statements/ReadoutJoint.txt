variable {Y : Type}

def weight (K : List Y → Y → ℝ) : List Y → List Y → ℝ
  | _, [] => 1
  | h, y :: rest => K h y * weight K (h ++ [y]) rest

def chain (eta : List Y → ℝ) : List Y → List Y → ℝ
  | _, [] => 1
  | h, y :: rest => eta h * chain eta (h ++ [y]) rest
