def Robust (y : ι → ℝ) (f : ℕ) (m : ℝ) : Prop :=
  f + 1 ≤ (univ.filter (fun i => y i ≤ m)).card ∧
    f + 1 ≤ (univ.filter (fun i => m ≤ y i)).card
