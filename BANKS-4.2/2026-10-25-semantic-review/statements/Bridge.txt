def ContinuesP {Y E Part : Type*} [NormedAddCommGroup E] (recY : Part → Y) (recE : Part → E)
    (ε : ℝ) (a : Part → Y × E) (x : Part) : Prop :=
  (a x).1 = recY x ∧ ‖(a x).2 - recE x‖ ≤ ε
