noncomputable def vecChan (v : ℝ≥0) :
    (n : ℕ) → (w : Fin n → α → ℝ) → (∀ i, Measurable (w i)) → Kernel α (Fin n → ℝ)
  | 0, _, _ => Kernel.const α (Measure.dirac (fun i => Fin.elim0 i))
  | n + 1, w, hw =>
    (chanW (w 0) (hw 0) v ×ₖ vecChan v n (fun i => w i.succ) (fun i => hw i.succ)).map
      (eSucc n).symm
