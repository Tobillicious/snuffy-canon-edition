def dynamicSurvival (t : ℝ) : ℝ := Real.exp (-t-t^2/2)
def dynamicReward (discount t : ℝ) : ℝ :=
  Real.exp (-discount*t)*dynamicSurvival t

  ∫ t in (0:ℝ)..horizon, dynamicReward discount t
